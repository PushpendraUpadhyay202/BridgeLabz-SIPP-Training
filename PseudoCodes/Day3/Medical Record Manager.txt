Interface Searchable :
    Method searchById(id)

Interface Printable :
    Method printReport()

//------------------------------------------------------------------------

Abstract Class MedicalReport implements Searchable, Printable :
    Attributes :
        INTEGER reportId
        STRING patientName

    Constructor(id, name) :
        SET self.reportId = id
        SET self.patientName = name

    Abstract Method generateReportContent()

    Method printReport() :
        PRINT "---------- Medical Report ----------"
        PRINT "Report ID: ", self.reportId
        PRINT "Patient: ", self.patientName
        CALL self.generateReportContent()
        PRINT "------------------------------------------"

    Method searchById(id) :
        IF self.reportId == id THEN
            RETURN self
        ELSE
            RETURN null
        ENDIF

//------------------------------------------------------------------------

Class BloodTestReport inherits MedicalReport :
    Attributes :
        FLOAT hemoglobinLevel
        INTEGER wbcCount

    Constructor(id, name, hemoglobin, wbc) :
        CALL super(id, name)
        SET self.hemoglobinLevel = hemoglobin
        SET self.wbcCount = wbc

    Method generateReportContent() :
        PRINT "Report Type: Blood Test"
        PRINT "Hemoglobin: ", self.hemoglobinLevel
        PRINT "WBC Count: ", self.wbcCount

//------------------------------------------------------------------------

Class XRayReport inherits MedicalReport :
    Attributes :
        STRING bodyPart
        STRING findings

    Constructor(id, name, part, findings) :
        CALL super(id, name)
        SET self.bodyPart = part
        SET self.findings = findings

    Method generateReportContent() :
        PRINT "Report Type: X-Ray"
        PRINT "Body Part Scanned: ", self.bodyPart
        PRINT "Findings: ", self.findings

//------------------------------------------------------------------------

Create report2 = XRayReport(1234, "Pushpendra", "Left Femur", "Hairline fracture found")
Create report1 = BloodTestReport(1008, "Arihant Jain", 14.5, 8500)

Create List<MedicalReport> reports = [report1, report2]

FOR EACH report IN reports :
    CALL report.printReport()
ENDFOR
